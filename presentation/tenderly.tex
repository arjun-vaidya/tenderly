% ---------- AI: how matching works ----------
\begin{frame}{How Matching Works}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Weighted Match Score}
        \vspace{0.05cm}
        \[
          \text{score} = \underbrace{0.45}_{\text{skills}} + \underbrace{0.25}_{\text{causes}} + \underbrace{0.20}_{\text{avail.}} + \underbrace{0.10}_{\text{location}}
        \]
        \vspace{-0.15cm}
      \end{block}
      \vspace{0.1cm}
      {\small
      \textbf{Skills} --- do your abilities match the org's needs?\\[0.06cm]
      \textbf{Causes} --- do you care about the same issues?\\[0.06cm]
      \textbf{Availability} --- can you show up when needed?\\[0.06cm]
      \textbf{Location} --- how active is need nearby?
      }
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{What Happens in a Cold Snap?}
        \begin{itemize}\setlength\itemsep{0pt}
          \item {\small Food \& shelter roles get a \textbf{score boost}}
          \item {\small Higher-need neighborhoods rise in ranking}
          \item {\small Urgency labels update (medium $\rightarrow$ high)}
        \end{itemize}
      \end{block}
      \vspace{0.1cm}
      \begin{alertblock}{Explainable, Not a Black Box}
        {\small Every match includes a plain-language reason so the volunteer knows \emph{why}.}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

        db/.style={flowstep, draw=tenderlyTerra, fill=tenderlyTerra!8},
        arr/.style={-{Stealth[length=4pt]}, tenderlyForest, line width=0.7pt},
      ]
        \node[flowstep] (src) {\textbf{Fetch from Partners}\\{\tiny nonprofit websites \& APIs}};
        \node[flowstep, below=of src] (hash) {\textbf{Check for Changes}\\{\tiny skip if content is unchanged}};
        \node[flowstep, below=of hash] (parse) {\textbf{AI Extracts Opportunities}\\{\tiny title, skills, location, schedule}};
        \node[flowstep, below=of parse] (score) {\textbf{Confidence Scoring}\\{\tiny how reliable is this listing?}};
        \node[db, below=of score] (approve) {\textbf{High confidence $\rightarrow$ Live}\\{\tiny low confidence $\rightarrow$ held for review}};

        \draw[arr] (src) -- (hash);
        \draw[arr] (hash) -- (parse);
        \draw[arr] (parse) -- (score);
        \draw[arr] (score) -- (approve);
      \end{tikzpicture}
      \end{center}
    \end{column}
    \begin{column}{0.50\textwidth}
      \begin{block}{Runs Every 6 Hours}
        {\small A scheduled job pulls the latest listings from partner nonprofits automatically.}
      \end{block}
      \vspace{0.15cm}
      \begin{block}{Smart Deduplication}
        {\small Each import is fingerprinted. Unchanged pages are skipped. Changed parsing rules trigger re-evaluation.}
      \end{block}
      \vspace{0.15cm}
      \begin{alertblock}{Source Isolation}
        {\small If one partner's site goes down, all other sources still import normally.}
      \end{alertblock}
    \end{column}
  \end{columns}
